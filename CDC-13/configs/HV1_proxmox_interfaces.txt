# CDC-13 - /etc/network/interfaces de l'hyperviseur HV1 (Proxmox VE) - CONCEPTION, NON TESTEE
# Agregat LACP (802.3ad) de 2 cartes + commutateur virtuel sensible aux VLAN
auto eno1
iface eno1 inet manual
auto eno2
iface eno2 inet manual

auto bond0
iface bond0 inet manual
    bond-slaves eno1 eno2
    bond-miimon 100
    bond-mode 802.3ad
    bond-xmit-hash-policy layer2+3

auto vmbr0
iface vmbr0 inet manual
    bridge-ports bond0
    bridge-stp off
    bridge-fd 0
    bridge-vlan-aware yes
    bridge-vids 60 70 99

# Gestion de l'hyperviseur dans le VLAN 99
auto vmbr0.99
iface vmbr0.99 inet static
    address 10.10.99.21/24
    gateway 10.10.99.1
# HV2 : identique, adresse 10.10.99.22/24
